            { emoji: '🎭', texto: 'A vida é um palco e você é o astro! Aproveite a cena!' }
        ];

        let contador = 0;

        function getRandomItem() {
            return coisasDivertidas[Math.floor(Math.random() * coisasDivertidas.length)];
        }

        function mostrarCoisa() {
            const item = getRandomItem();
            document.getElementById('emoji').textContent = item.emoji;
            document.getElementById('content').textContent = item.texto;
            contador++;
            atualizarStats();
        }

        function mostrarTudo() {
            let html = '<div style="text-align: left;">';
            coisasDivertidas.forEach((item, index) => {
                html += `<p style="margin: 10px 0;"><strong>${item.emoji} ${index + 1}.</strong> ${item.texto}</p>`;
            });
            html += '</div>';
            document.getElementById('content').innerHTML = html;
            contador += coisasDivertidas.length;
            atualizarStats();
        }

        function atualizarStats() {
            document.getElementById('stats').textContent = `Você já viu ${contador} coisa(s) divertida(s)! 🎯`;
        }

        // Mostrar algo divertido ao carregar
        window.addEventListener('load', mostrarCoisa);
